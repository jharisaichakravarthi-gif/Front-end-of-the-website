11. LANGUAGE CONTROLLER IN JAVASCRIPT CODE:

let currentLanguage =
    localStorage.getItem("agriShieldLanguage") || "en";


function translatePage() {

    const dictionary =
        translations[currentLanguage] ||
        translations.en;


    document.documentElement.lang =
        currentLanguage;


    document
        .querySelectorAll("[data-i18n]")
        .forEach(element => {

            const key =
                element.dataset.i18n;

            if (dictionary[key]) {

                element.textContent =
                    dictionary[key];

            }

        });


    document
        .querySelectorAll("[data-i18n-placeholder]")
        .forEach(element => {

            const key =
                element.dataset.i18nPlaceholder;

            if (dictionary[key]) {

                element.placeholder =
                    dictionary[key];

            }

        });


    updateNavigationLanguage();

}


function changeLanguage(language) {

    if (!translations[language]) {

        language = "en";

    }


    currentLanguage = language;


    localStorage.setItem(
        "agriShieldLanguage",
        language
    );


    translatePage();

}


function updateNavigationLanguage() {

    document
        .querySelectorAll(".nav button")
        .forEach(button => {

            const page =
                button.dataset.page;

            const span =
                button.querySelector("[data-i18n]");

            if (
                span &&
                translations[currentLanguage][page]
            ) {

                span.textContent =
                    translations[currentLanguage][page];

            }

        });

}


document
    .getElementById("language")
    ?.addEventListener(
        "change",
        event => {

            changeLanguage(
                event.target.value
            );

        }
    );


document.addEventListener(
    "DOMContentLoaded",
    () => {

        const selector =
            document.getElementById("language");

        if (selector) {

            selector.value =
                currentLanguage;

        }

        translatePage();

    }
);

12. MAIN APPLICATION IN JAWASCRIPT:

const state = {

    currentPage: "dashboard",

    crops: [],

    alerts: [],

    irrigation: {

        pump: false,

        moisture: 28,

        schedule: null

    },

    profile: {

        name: "",

        phone: "",

        email: ""

    }

};


function go(page) {

    const pages =
        document.querySelectorAll(".page");

    pages.forEach(section => {

        section.classList.remove("active");

    });


    const target =
        document.getElementById(page);

    if (!target) return;


    target.classList.add("active");


    document
        .querySelectorAll(".nav button")
        .forEach(button => {

            button.classList.remove("active");

        });


    const nav =
        document.querySelector(
            `.nav button[data-page="${page}"]`
        );


    if (nav) {

        nav.classList.add("active");

    }


    state.currentPage = page;


    updatePageTitle(page);


    window.scrollTo({

        top: 0,

        behavior: "smooth"

    });


    translatePage();

}


function updatePageTitle(page) {

    const title =
        document.getElementById("pageTitle");

    if (!title) return;


    const dictionary =
        translations[currentLanguage];


    if (dictionary[page]) {

        title.textContent =
            dictionary[page];

    }

}


function addCrop() {

    const name =
        document.getElementById("cropName")
            ?.value.trim();

    const area =
        document.getElementById("cropArea")
            ?.value.trim();


    if (!name) {

        alert("Please enter crop name.");

        return;

    }


    state.crops.push({

        id: Date.now(),

        name,

        area

    });


    saveState();

    renderCrops();

}


function renderCrops() {

    const container =
        document.getElementById("cropList");

    if (!container) return;


    container.innerHTML = "";


    state.crops.forEach(crop => {

        const card =
            document.createElement("div");

        card.className =
            "crop-card";


        card.innerHTML = `

            <div class="crop-icon">
                🌱
            </div>

            <div>

                <h3>
                    ${escapeHTML(crop.name)}
                </h3>

                <p>
                    ${escapeHTML(crop.area || "")}
                </p>

            </div>

        `;


        container.appendChild(card);

    });


    const count =
        document.getElementById("cropCount");


    if (count) {

        count.textContent =
            state.crops.length;

    }

}


function togglePump() {

    state.irrigation.pump =
        !state.irrigation.pump;


    const status =
        document.getElementById("pumpStatus");


    if (status) {

        status.textContent =
            state.irrigation.pump
                ? "ON"
                : "OFF";

    }


    saveState();

}


function saveIrrigationSchedule() {

    const time =
        document.getElementById(
            "irrigationTime"
        )?.value;


    state.irrigation.schedule =
        time;


    saveState();

}


function saveProfile() {

    state.profile.name =
        document.getElementById(
            "profileName"
        )?.value || "";


    state.profile.phone =
        document.getElementById(
            "profilePhone"
        )?.value || "";


    state.profile.email =
        document.getElementById(
            "profileEmail"
        )?.value || "";


    saveState();

}


function saveState() {

    localStorage.setItem(

        "agriShieldState",

        JSON.stringify(state)

    );

}


function loadState() {

    try {

        const saved =
            localStorage.getItem(
                "agriShieldState"
            );


        if (saved) {

            Object.assign(
                state,
                JSON.parse(saved)
            );

        }

    }

    catch(error) {

        console.error(
            "State loading failed",
            error
        );

    }

}


function escapeHTML(value) {

    return String(value)

        .replaceAll("&", "&amp;")

        .replaceAll("<", "&lt;")

        .replaceAll(">", "&gt;")

        .replaceAll('"', "&quot;")

        .replaceAll("'", "&#039;");

}

const pesticides = [

    {
        name: "Neem Oil",
        category: "biological",

    {
        name: "Azadirachtin",
        category: "biological",
        crop: "Vegetables"

    {

        category: "biological",
        crop: "Multiple Crops"
    },

    {
        name: "Copper Fungicide",
        category: "fungicide",
        crop: "Vegetables"
    },

    {
        name: "Sulfur",
        category: "fungicide",
        crop: "Multiple Crops"
    }

];


function renderPesticides() {

    const container =
        document.getElementById(
            "pesticideList"
        );


    if (!container) return;


    const search =
        document.getElementById(
            "pesticideSearch"
        )?.value
            .toLowerCase() || "";


    const category =
        document.getElementById(
            "pesticideCategory"
        )?.value || "all";


    const results =
        pesticides.filter(product => {

            const matchesSearch =
                product.name
                    .toLowerCase()
                    .includes(search);


            const matchesCategory =
                category === "all" ||
                product.category === category;


            return (
                matchesSearch &&
                matchesCategory
            );

        });


    container.innerHTML = "";


    results.forEach(product => {

        const card =
            document.createElement("div");

        card.className =
            "card pesticide-card";


        card.innerHTML = `

            <div class="pesticide-icon">
                🧪
            </div>

            <h3>
                ${escapeHTML(product.name)}
            </h3>

            <p>
                ${escapeHTML(product.crop)}
            </p>

            <span class="tag">
                ${escapeHTML(product.category)}
            </span>

        `;


        container.appendChild(card);

    });

}


document.addEventListener(

    "DOMContentLoaded",

    () => {

        document
            .getElementById(
                "pesticideSearch"
            )
            ?.addEventListener(
                "input",
                renderPesticides
            );


        document
            .getElementById(
                "pesticideCategory"
            )
            ?.addEventListener(
                "change",
                renderPesticides
            );


        renderPesticides();

    }

);
